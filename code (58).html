# قاعدة البيانات (PostgreSQL / Supabase Schema): 
-- 1. جدول الوظائف الشاغرة
CREATE TABLE jobs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  employer_id UUID REFERENCES employers(id),
  title VARCHAR(255) NOT NULL,
  category VARCHAR(100), -- 'صحة وصيدلة', 'إدارة ومحاسبة', 'تقنية'
  city VARCHAR(50) DEFAULT 'إب',
  district VARCHAR(100), -- 'الدائري', 'شارع العدين', 'جبلة'
  salary_yer NUMERIC(12, 2), -- مثل 250,000 ريال يمني
  housing_provided BOOLEAN DEFAULT false,
  is_urgent BOOLEAN DEFAULT false,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- 2. جدول المدفوعات بمحفظة جيب
CREATE TABLE payments (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL,
  wallet_type VARCHAR(50) DEFAULT 'jaib_wallet',
  jaib_account_target VARCHAR(20) DEFAULT '770088836',
  amount_yer NUMERIC(12,2) NOT NULL,
  tx_reference VARCHAR(100) UNIQUE,
  status VARCHAR(20) DEFAULT 'pending' -- 'verified', 'approved'
);
# المسارات البرمجية المتاحة (RESTful Endpoints):
GET /api/v1/jobs?governorate=ibb&urgent=true جلب شواغر محافظة إب العاجلة
